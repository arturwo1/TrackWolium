if(WIN32)
  set(TW_PLATFORM_SRC windows/win_probe.cpp windows/paths_win.cpp)
elseif(APPLE)
  set(TW_PLATFORM_SRC macos/mac_probe.mm linux/paths_posix.cpp)
  set_source_files_properties(macos/mac_probe.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
else()
  set(TW_PLATFORM_SRC linux/linux_probe.cpp linux/x11_windows.cpp linux/paths_posix.cpp)
endif()

add_library(tw_platform STATIC ${TW_PLATFORM_SRC})
target_include_directories(tw_platform PUBLIC include)
target_link_libraries(tw_platform PUBLIC tw_core)
if(WIN32)
  target_link_libraries(tw_platform PRIVATE dwmapi version)
endif()
if(APPLE)
  target_link_libraries(tw_platform PRIVATE "-framework AppKit" "-framework CoreGraphics" "-framework Foundation")
endif()
if(NOT WIN32 AND NOT APPLE)
  find_path(TW_X11_INCLUDE X11/Xlib.h)
  if(TW_X11_INCLUDE)
    target_compile_definitions(tw_platform PRIVATE TW_HAVE_X11)
    target_include_directories(tw_platform PRIVATE ${TW_X11_INCLUDE})
  endif()
  target_link_libraries(tw_platform PRIVATE ${CMAKE_DL_LIBS})
endif()
