foreach(tool setup uninstall)
  add_executable(wolium_${tool} WIN32 ${tool}_win.cpp)
  target_link_libraries(wolium_${tool} PRIVATE ole32 uuid shell32 user32 gdi32 advapi32)
  if(MSVC)
    target_link_options(wolium_${tool} PRIVATE /ENTRY:wWinMainCRTStartup)
  else()
    target_link_options(wolium_${tool} PRIVATE -municode -static)
  endif()
endforeach()
set_target_properties(wolium_setup PROPERTIES OUTPUT_NAME WoliumSetup)
set_target_properties(wolium_uninstall PROPERTIES OUTPUT_NAME uninstall)
